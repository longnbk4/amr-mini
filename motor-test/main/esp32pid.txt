#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "driver/gpio.h"
#include "driver/ledc.h"
#include "driver/uart.h"
#include "esp_log.h"

// ==================== CHÂN PHẦN CỨNG ====================
#define L_PWM_PIN       10
#define L_IN1_PIN       12
#define L_IN2_PIN       13
#define R_PWM_PIN       16
#define R_IN1_PIN       14
#define R_IN2_PIN       15

#define L_ENC_B_PIN     6
#define L_ENC_A_PIN     7
#define R_ENC_A_PIN     8
#define R_ENC_B_PIN     9

#define UART_TX_PIN     17
#define UART_RX_PIN     18
#define UART_PORT_NUM   UART_NUM_1
#define BUF_SIZE        1024
#define COMMAND_TIMEOUT_MS 500

volatile long left_ticks  = 0;
volatile long right_ticks = 0;
static volatile uint32_t last_command_time = 0;
static const char *TAG = "UART_PID_BRIDGE";

// ==================== THÔNG SỐ PID ====================
float Kp = 2.5; 
float Ki = 0.1;
float Kd = 0.05;

// Vận tốc mục tiêu (Setpoint: Ticks/50ms)
static int target_left = 0;
static int target_right = 0;

// ==================== ISR ENCODER ====================
static void IRAM_ATTR left_enc_isr_handler(void* arg) {
    if (gpio_get_level(L_ENC_B_PIN)) left_ticks++;
    else left_ticks--;
}

static void IRAM_ATTR right_enc_isr_handler(void* arg) {
    if (gpio_get_level(R_ENC_B_PIN)) right_ticks++;
    else right_ticks--;
}

// ==================== KHỞI TẠO PHẦN CỨNG ====================
void init_hardware(void) {
    ledc_timer_config_t timer = {
        .speed_mode = LEDC_LOW_SPEED_MODE,
        .timer_num = LEDC_TIMER_0,
        .duty_resolution = LEDC_TIMER_8_BIT,
        .freq_hz = 5000,
        .clk_cfg = LEDC_AUTO_CLK
    };
    ledc_timer_config(&timer);

    ledc_channel_config_t ch_l = {
        .speed_mode = LEDC_LOW_SPEED_MODE, .channel = LEDC_CHANNEL_0,
        .timer_sel = LEDC_TIMER_0, .intr_type = LEDC_INTR_DISABLE,
        .gpio_num = L_PWM_PIN, .duty = 0, .hpoint = 0
    };
    ledc_channel_config_t ch_r = {
        .speed_mode = LEDC_LOW_SPEED_MODE, .channel = LEDC_CHANNEL_1,
        .timer_sel = LEDC_TIMER_0, .intr_type = LEDC_INTR_DISABLE,
        .gpio_num = R_PWM_PIN, .duty = 0, .hpoint = 0
    };
    ledc_channel_config(&ch_l);
    ledc_channel_config(&ch_r);

    gpio_set_direction(L_IN1_PIN, GPIO_MODE_OUTPUT);
    gpio_set_direction(L_IN2_PIN, GPIO_MODE_OUTPUT);
    gpio_set_direction(R_IN1_PIN, GPIO_MODE_OUTPUT);
    gpio_set_direction(R_IN2_PIN, GPIO_MODE_OUTPUT);

    gpio_config_t io = {
        .intr_type = GPIO_INTR_POSEDGE,
        .mode = GPIO_MODE_INPUT,
        .pin_bit_mask = (1ULL << L_ENC_A_PIN) | (1ULL << R_ENC_A_PIN),
        .pull_up_en = 1
    };
    gpio_config(&io);

    io.intr_type = GPIO_INTR_DISABLE;
    io.pin_bit_mask = (1ULL << L_ENC_B_PIN) | (1ULL << R_ENC_B_PIN);
    gpio_config(&io);

    gpio_install_isr_service(0);
    gpio_isr_handler_add(L_ENC_A_PIN, left_enc_isr_handler, NULL);
    gpio_isr_handler_add(R_ENC_A_PIN, right_enc_isr_handler, NULL);

    uart_config_t uart_config = {
        .baud_rate = 115200,
        .data_bits = UART_DATA_8_BITS,
        .parity = UART_PARITY_DISABLE,
        .stop_bits = UART_STOP_BITS_1,
        .flow_ctrl = UART_HW_FLOWCTRL_DISABLE,
        .source_clk = UART_SCLK_DEFAULT
    };
    uart_driver_install(UART_PORT_NUM, BUF_SIZE * 2, 0, 0, NULL, 0);
    uart_param_config(UART_PORT_NUM, &uart_config);
    uart_set_pin(UART_PORT_NUM, UART_TX_PIN, UART_RX_PIN, UART_PIN_NO_CHANGE, UART_PIN_NO_CHANGE);
}

// ==================== XUẤT PWM L298N ====================
void apply_pwm(int left_pwm, int right_pwm) {
    if (left_pwm >= 0) {
        gpio_set_level(L_IN1_PIN, 1);
        gpio_set_level(L_IN2_PIN, 0);
        ledc_set_duty(LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_0, left_pwm);
    } else {
        gpio_set_level(L_IN1_PIN, 0);
        gpio_set_level(L_IN2_PIN, 1);
        ledc_set_duty(LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_0, -left_pwm);
    }
    ledc_update_duty(LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_0);

    if (right_pwm >= 0) {
        gpio_set_level(R_IN1_PIN, 1);
        gpio_set_level(R_IN2_PIN, 0);
        ledc_set_duty(LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_1, right_pwm);
    } else {
        gpio_set_level(R_IN1_PIN, 0);
        gpio_set_level(R_IN2_PIN, 1);
        ledc_set_duty(LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_1, -right_pwm);
    }
    ledc_update_duty(LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_1);
}

// ==================== TASK PID ĐIỀU KHIỂN VẬN TỐC ====================
static void pid_control_task(void *arg) {
    long prev_left_ticks = 0, prev_right_ticks = 0;
    float integral_L = 0, integral_R = 0;
    float prev_speed_L = 0, prev_speed_R = 0;   // Dùng cho Derivative on Measurement

    while (1) {
        long curr_L = left_ticks;
        long curr_R = right_ticks;

        // Tốc độ thực tế (tick / 50ms)
        int speed_L = curr_L - prev_left_ticks;
        int speed_R = curr_R - prev_right_ticks;
        prev_left_ticks = curr_L;
        prev_right_ticks = curr_R;

        if (target_left == 0 && target_right == 0) {
            integral_L = integral_R = 0;
            prev_speed_L = prev_speed_R = 0;
            apply_pwm(0, 0);
        } else {
            // --- LEFT ---
            float error_L = target_left - speed_L;
            integral_L += error_L;
            // Anti-windup
            if (integral_L > 400) integral_L = 400;
            if (integral_L < -400) integral_L = -400;

            float derivative_L = -(speed_L - prev_speed_L);  // Derivative on Measurement
            prev_speed_L = speed_L;

            float pwm_L = Kp * error_L + Ki * integral_L + Kd * derivative_L;

            // --- RIGHT ---
            float error_R = target_right - speed_R;
            integral_R += error_R;
            // Anti-windup
            if (integral_R > 400) integral_R = 400;
            if (integral_R < -400) integral_R = -400;

            float derivative_R = -(speed_R - prev_speed_R);
            prev_speed_R = speed_R;

            float pwm_R = Kp * error_R + Ki * integral_R + Kd * derivative_R;

            // Giới hạn PWM
            if (pwm_L > 255) pwm_L = 255;
            if (pwm_L < -255) pwm_L = -255;
            if (pwm_R > 255) pwm_R = 255;
            if (pwm_R < -255) pwm_R = -255;

            apply_pwm((int)pwm_L, (int)pwm_R);
        }

        vTaskDelay(pdMS_TO_TICKS(50));
    }
}

// ==================== TASK NHẬN LỆNH UART TỪ PI 4 ====================
static void uart_rx_task(void *arg) {
    uint8_t *data = malloc(BUF_SIZE);
    char line[128];
    int pos = 0;

    while (1) {
        int len = uart_read_bytes(UART_PORT_NUM, data, BUF_SIZE - 1, pdMS_TO_TICKS(20));
        if (len > 0) {
            for (int i = 0; i < len; i++) {
                char c = data[i];
                if (c == '\n' || c == '\r') {
                    if (pos > 0) {
                        line[pos] = 0;
                        // Lệnh từ Pi 4 có dạng: V,<target_left>,<target_right>
                        if (line[0] == 'V') {
                            int l = 0, r = 0;
                            if (sscanf(line, "V,%d,%d", &l, &r) == 2) {
                                target_left = l;   // Cập nhật mục tiêu PID
                                target_right = r;  // Cập nhật mục tiêu PID
                                last_command_time = xTaskGetTickCount() * portTICK_PERIOD_MS;
                                // Không cần phải write lại "OK\n" nữa để tránh rác cho luồng Python
                            }
                        }
                        pos = 0;
                    }
                } else if (pos < sizeof(line) - 1) {
                    line[pos++] = c;
                }
            }
        }

        // Timeout an toàn: Mất kết nối Pi4 -> Dừng xe
        if ((xTaskGetTickCount() * portTICK_PERIOD_MS - last_command_time) > COMMAND_TIMEOUT_MS) {
            target_left = 0;
            target_right = 0;
        }

        vTaskDelay(pdMS_TO_TICKS(5));
    }
    free(data);
}

// ==================== MAIN ====================
void app_main(void) {
    init_hardware();
    target_left = 0;
    target_right = 0;
    last_command_time = xTaskGetTickCount() * portTICK_PERIOD_MS;

    // Chạy 2 Task: Nhận lệnh và Chạy PID
    xTaskCreate(uart_rx_task, "uart_rx", 4096, NULL, 10, NULL);
    xTaskCreate(pid_control_task, "pid_task", 4096, NULL, 6, NULL);

    ESP_LOGI(TAG, "=== UART BRIDGE (WITH PID) READY ===");
    ESP_LOGI(TAG, "TX=%d RX=%d | Baud=115200", UART_TX_PIN, UART_RX_PIN);

    char tx_buf[64];
    while (1) {
        // Gửi Odometry về cho Pi 4 ở tần số 20 Hz
        int len = snprintf(tx_buf, sizeof(tx_buf), "E,%ld,%ld\n", left_ticks, right_ticks);
        uart_write_bytes(UART_PORT_NUM, tx_buf, len);
        vTaskDelay(pdMS_TO_TICKS(50));   
    }
}
