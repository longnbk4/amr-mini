// 1. ĐỊNH NGHĨA CHÂN L298N
#define L_PWM_PIN 10
#define L_IN1_PIN 12
#define L_IN2_PIN 13
#define R_PWM_PIN 16
#define R_IN1_PIN 14
#define R_IN2_PIN 15

// 2. ĐỊNH NGHĨA CHÂN ENCODER
#define L_ENC_B_PIN 6
#define L_ENC_A_PIN 7
#define R_ENC_A_PIN 8
#define R_ENC_B_PIN 9

volatile long left_ticks = 0;
volatile long right_ticks = 0;

// Biến trạng thái điều khiển
static int motor_speed = 30; // Tốc độ mục tiêu (Target ticks / 50ms)
static volatile uint32_t last_command_time = 0;
static volatile bool robot_moving = false;
static const uint32_t COMMAND_TIMEOUT_MS = 400;
static const char *TAG = "ROBOT_PID";

// ==========================================
// THÔNG SỐ PID 
// ==========================================
float Kp = 2.5; 
float Ki = 0.1;
float Kd = 0.3;

// Vận tốc mục tiêu (Setpoint) cho từng bánh
static int target_left = 0;
static int target_right = 0;

// ==========================================
// ISR Encoder
// ==========================================
static void IRAM_ATTR left_enc_isr_handler(void *arg) {
    if (gpio_get_level(L_ENC_B_PIN)) left_ticks++;
    else left_ticks--;
}

static void IRAM_ATTR right_enc_isr_handler(void* arg) {
    if (gpio_get_level(R_ENC_B_PIN)) right_ticks++;
    else right_ticks--;
}

// ==========================================
// Khởi tạo phần cứng
// ==========================================
void init_hardware(void) {
    ledc_timer_config_t ledc_timer = {
        .speed_mode = LEDC_LOW_SPEED_MODE,
        .timer_num  = LEDC_TIMER_0,
        .duty_resolution = LEDC_TIMER_8_BIT,
        .freq_hz    = 5000,
        .clk_cfg    = LEDC_AUTO_CLK
    };
    ledc_timer_config(&ledc_timer);

    ledc_channel_config_t ledc_channel_L = {
        .speed_mode = LEDC_LOW_SPEED_MODE,
        .channel    = LEDC_CHANNEL_0,
        .timer_sel  = LEDC_TIMER_0,
        .intr_type  = LEDC_INTR_DISABLE,
        .gpio_num   = L_PWM_PIN,
        .duty       = 0,
        .hpoint     = 0
    };
    ledc_channel_config(&ledc_channel_L);

    ledc_channel_config_t ledc_channel_R = {
        .speed_mode = LEDC_LOW_SPEED_MODE,
        .channel    = LEDC_CHANNEL_1,
        .timer_sel  = LEDC_TIMER_0,
        .intr_type  = LEDC_INTR_DISABLE,
        .gpio_num   = R_PWM_PIN,
        .duty       = 0,
        .hpoint     = 0
    };
    ledc_channel_config(&ledc_channel_R);

    gpio_set_direction(L_IN1_PIN, GPIO_MODE_OUTPUT);
    gpio_set_direction(L_IN2_PIN, GPIO_MODE_OUTPUT);
    gpio_set_direction(R_IN1_PIN, GPIO_MODE_OUTPUT);
    gpio_set_direction(R_IN2_PIN, GPIO_MODE_OUTPUT);

    gpio_config_t io_conf = {};
    io_conf.intr_type = GPIO_INTR_POSEDGE;
    io_conf.pin_bit_mask = (1ULL << L_ENC_A_PIN) | (1ULL << R_ENC_A_PIN);
    io_conf.mode = GPIO_MODE_INPUT;
    io_conf.pull_up_en = 1;
    gpio_config(&io_conf);

    io_conf.intr_type = GPIO_INTR_DISABLE;
    io_conf.pin_bit_mask = (1ULL << L_ENC_B_PIN) | (1ULL << R_ENC_B_PIN);
    io_conf.pull_up_en = 1;
    gpio_config(&io_conf);

    gpio_install_isr_service(0);
    gpio_isr_handler_add(L_ENC_A_PIN, left_enc_isr_handler, NULL);
    gpio_isr_handler_add(R_ENC_A_PIN, right_enc_isr_handler, NULL);
}

// ==========================================
// Cấp trực tiếp PWM ra chân L298N
// ==========================================
void apply_pwm(int left_pwm, int right_pwm) {
    if (left_pwm > 255) left_pwm = 255;
    if (left_pwm < -255) left_pwm = -255;
    if (right_pwm > 255) right_pwm = 255;
    if (right_pwm < -255) right_pwm = -255;

    // Motor Trái
    if (left_pwm >= 0) {
        gpio_set_level(L_IN1_PIN, 1);
        gpio_set_level(L_IN2_PIN, 0);
        ledc_set_duty(LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_0, left_pwm);
    } else {
        gpio_set_level(L_IN1_PIN, 0);
        gpio_set_level(L_IN2_PIN, 1);
        ledc_set_duty(LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_0, -left_pwm);
    }
    ledc_update_duty(LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_0);

    // Motor Phải
    if (right_pwm >= 0) {
        gpio_set_level(R_IN1_PIN, 1);
        gpio_set_level(R_IN2_PIN, 0);
        ledc_set_duty(LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_1, right_pwm);
    } else {
        gpio_set_level(R_IN1_PIN, 0);
        gpio_set_level(R_IN2_PIN, 1);
        ledc_set_duty(LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_1, -right_pwm);
    }
    ledc_update_duty(LEDC_LOW_SPEED_MODE, LEDC_CHANNEL_1);
}

// ==========================================
// TASK PID ĐIỀU KHIỂN VẬN TỐC
// ==========================================
static void pid_control_task(void *arg) {
    long prev_left_ticks = 0, prev_right_ticks = 0;
    float integral_L = 0, integral_R = 0;
    float prev_speed_L = 0, prev_speed_R = 0;   // Dùng cho Derivative on Measurement

    while (1) {
        long curr_L = left_ticks;
        long curr_R = right_ticks;

        // Tốc độ thực tế (tick / 50ms)
        int speed_L = curr_L - prev_left_ticks;
        int speed_R = curr_R - prev_right_ticks;
        
        prev_left_ticks = curr_L;
        prev_right_ticks = curr_R;

        if (target_left == 0 && target_right == 0) {
            integral_L = integral_R = 0;
            prev_speed_L = prev_speed_R = 0;
            apply_pwm(0, 0);
        } else {
            // --- BÁNH TRÁI ---
            float error_L = target_left - speed_L;
            integral_L += error_L;
            // Anti-windup
            if (integral_L > 400) integral_L = 400;
            if (integral_L < -400) integral_L = -400;

            float derivative_L = -(speed_L - prev_speed_L);  // Derivative on Measurement
            prev_speed_L = speed_L;

            float pwm_L = Kp * error_L + Ki * integral_L + Kd * derivative_L;

            // --- BÁNH PHẢI ---
            float error_R = target_right - speed_R;
            integral_R += error_R;
            // Anti-windup
            if (integral_R > 400) integral_R = 400;
            if (integral_R < -400) integral_R = -400;

            float derivative_R = -(speed_R - prev_speed_R);
            prev_speed_R = speed_R;

            float pwm_R = Kp * error_R + Ki * integral_R + Kd * derivative_R;

            // Giới hạn PWM bão hòa trước khi cấp xuống Driver
            if (pwm_L > 255) pwm_L = 255;
            if (pwm_L < -255) pwm_L = -255;
            if (pwm_R > 255) pwm_R = 255;
            if (pwm_R < -255) pwm_R = -255;

            apply_pwm((int)pwm_L, (int)pwm_R);
        }

        vTaskDelay(pdMS_TO_TICKS(50));
    }
}

// ==========================================
// CÁC HÀM XỬ LÝ LỆNH TỪ WEB
// ==========================================
void stop_robot(void) {
    target_left = 0;
    target_right = 0;
}

void execute_movement(const char *cmd) {
    robot_moving = true;
    last_command_time = xTaskGetTickCount() * portTICK_PERIOD_MS;
    
    // Thiết lập mục tiêu tốc độ (ticks/50ms)
    if (strcmp(cmd, "FORWARD") == 0) {
        target_left = motor_speed;
        target_right = motor_speed;
    } else if (strcmp(cmd, "BACKWARD") == 0) {
        target_left = -motor_speed;
        target_right = -motor_speed;
    } else if (strcmp(cmd, "LEFT") == 0) {
        target_left = -motor_speed;
        target_right = motor_speed;
    } else if (strcmp(cmd, "RIGHT") == 0) {
        target_left = motor_speed;
        target_right = -motor_speed;
    } else {
        stop_robot();
        robot_moving = false;
    }
}

// HTTP Handlers
// ==========================================
static esp_err_t root_get_handler(httpd_req_t *req) {
    httpd_resp_set_type(req, "text/html");
    httpd_resp_send(req, CONTROL_PAGE, HTTPD_RESP_USE_STRLEN);
    return ESP_OK;
}

static esp_err_t command_get_handler(httpd_req_t *req) {
    char buf[64] = {0};
    if (httpd_req_get_url_query_str(req, buf, sizeof(buf)) == ESP_OK) {
        char value[32] = {0};
        if (httpd_query_key_value(buf, "value", value, sizeof(value)) == ESP_OK) {
            execute_movement(value);
        }
    }
    httpd_resp_sendstr(req, "OK");
    return ESP_OK;
}

static esp_err_t speed_get_handler(httpd_req_t *req) {
    char buf[64] = {0};
    if (httpd_req_get_url_query_str(req, buf, sizeof(buf)) == ESP_OK) {
        char value[16] = {0};
        if (httpd_query_key_value(buf, "value", value, sizeof(value)) == ESP_OK) {
            int v = atoi(value);
            if (v < 10) v = 10;
            if (v > 150) v = 150;
            motor_speed = v;
            
            // Nếu robot đang chạy, cập nhật mục tiêu trực tiếp
            if (robot_moving && target_left != 0) {
                 target_left = (target_left > 0) ? motor_speed : -motor_speed;
                 target_right = (target_right > 0) ? motor_speed : -motor_speed;
            }
        }
    }
    char resp[16];
    snprintf(resp, sizeof(resp), "%d", motor_speed);
    httpd_resp_sendstr(req, resp);
    return ESP_OK;
}

// ==========================================
// Khởi tạo Wi-Fi SoftAP
// ==========================================
static void wifi_init_softap(void) {
    ESP_ERROR_CHECK(esp_netif_init());
    ESP_ERROR_CHECK(esp_event_loop_create_default());
    esp_netif_create_default_wifi_ap();

    wifi_init_config_t cfg = WIFI_INIT_CONFIG_DEFAULT();
    ESP_ERROR_CHECK(esp_wifi_init(&cfg));

    wifi_config_t wifi_config = {
        .ap = {
            .ssid = "JGA_Robot_PID",
            .ssid_len = 0,
            .password = "robot1234",
            .max_connection = 4,
            .authmode = WIFI_AUTH_WPA_WPA2_PSK
        },
    };
    ESP_ERROR_CHECK(esp_wifi_set_mode(WIFI_MODE_AP));
    ESP_ERROR_CHECK(esp_wifi_set_config(WIFI_IF_AP, &wifi_config));
    ESP_ERROR_CHECK(esp_wifi_start());

    ESP_LOGI(TAG, "SoftAP started -> SSID: JGA_Robot_PID | Pass: robot1234");
}

static httpd_handle_t start_webserver(void) {
    httpd_config_t config = HTTPD_DEFAULT_CONFIG();
    config.max_uri_handlers = 8;
    httpd_handle_t server = NULL;
    
    if (httpd_start(&server, &config) == ESP_OK) {
        httpd_uri_t root = { .uri = "/", .method = HTTP_GET, .handler = root_get_handler };
        httpd_register_uri_handler(server, &root);

        httpd_uri_t cmd = { .uri = "/command", .method = HTTP_GET, .handler = command_get_handler };
        httpd_register_uri_handler(server, &cmd);

        httpd_uri_t spd = { .uri = "/speed", .method = HTTP_GET, .handler = speed_get_handler };
        httpd_register_uri_handler(server, &spd);
    }
    return server;
}

// Task timeout an toàn
static void safety_task(void *arg) {
    while (1) {
        if (robot_moving) {
            uint32_t now = xTaskGetTickCount() * portTICK_PERIOD_MS;
            if (now - last_command_time > COMMAND_TIMEOUT_MS) {
                ESP_LOGW(TAG, "Timeout -> STOP");
                stop_robot();
                robot_moving = false;
            }
        }
        vTaskDelay(pdMS_TO_TICKS(50));
    }
}

// ==========================================
// MAIN
// ==========================================
void app_main(void) {
    esp_err_t ret = nvs_flash_init();
    if (ret == ESP_ERR_NVS_NO_FREE_PAGES || ret == ESP_ERR_NVS_NEW_VERSION_FOUND) {
        ESP_ERROR_CHECK(nvs_flash_erase());
        ret = nvs_flash_init();
    }
    ESP_ERROR_CHECK(ret);

    init_hardware();
    stop_robot();

    wifi_init_softap();
    start_webserver();

    // Khởi tạo các Task
    xTaskCreate(safety_task, "safety", 2048, NULL, 5, NULL);
    xTaskCreate(pid_control_task, "pid_task", 4096, NULL, 6, NULL); // Task PID chạy ở Priority 6

    ESP_LOGI(TAG, "=== ROBOT READY (PID ENABLED) ===");
    ESP_LOGI(TAG, "Truy cap: http://192.168.4.1");

    // In log giám sát PID mỗi giây
    while (1) {
        ESP_LOGI(TAG, "Target L/R: %d/%d | Ticks L: %ld | Ticks R: %ld",
                 target_left, target_right, left_ticks, right_ticks);
        vTaskDelay(pdMS_TO_TICKS(1000));
    }
}
